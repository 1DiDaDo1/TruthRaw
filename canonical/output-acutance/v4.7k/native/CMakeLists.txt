cmake_minimum_required(VERSION 3.16)
project(truthraw_output_acutance_v47k LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(truthraw_output_acutance STATIC output_acutance.cpp)
target_include_directories(truthraw_output_acutance PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(truthraw_output_acutance PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(truthraw_acutance_cli acutance_cli.cpp)
target_link_libraries(truthraw_acutance_cli PRIVATE truthraw_output_acutance)
target_compile_options(truthraw_acutance_cli PRIVATE -Wall -Wextra -Wpedantic -Werror)

add_executable(truthraw_acutance_bench bench_acutance.cpp)
target_link_libraries(truthraw_acutance_bench PRIVATE truthraw_output_acutance)
target_compile_options(truthraw_acutance_bench PRIVATE -Wall -Wextra -Wpedantic -Werror)
