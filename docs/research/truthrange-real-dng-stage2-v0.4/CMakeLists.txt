cmake_minimum_required(VERSION 3.16)
project(truthraw_real_dng_stage2_v0_4 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(truthraw_dng_stage2_v0_4 native/dng_stage2_v0_4.cpp)
target_include_directories(truthraw_dng_stage2_v0_4 PUBLIC native)
add_executable(test_dng_stage2_v0_4 tests/test_dng_stage2_v0_4.cpp)
target_link_libraries(test_dng_stage2_v0_4 PRIVATE truthraw_dng_stage2_v0_4)
add_executable(dng_stage2_v0_4_cli native/dng_stage2_v0_4_cli.cpp)
target_link_libraries(dng_stage2_v0_4_cli PRIVATE truthraw_dng_stage2_v0_4)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  foreach(t truthraw_dng_stage2_v0_4 test_dng_stage2_v0_4 dng_stage2_v0_4_cli)
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endforeach()
endif()
