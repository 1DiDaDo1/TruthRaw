cmake_minimum_required(VERSION 3.16)
project(truthraw_uncertainty_aware_scurve_color_v1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(uncertainty_aware_scurve_color_v1 native/uncertainty_aware_scurve_color_v1.cpp)
target_include_directories(uncertainty_aware_scurve_color_v1 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/native)
add_executable(test_uncertainty_aware_scurve_color_v1 tests/test_uncertainty_aware_scurve_color_v1.cpp)
target_link_libraries(test_uncertainty_aware_scurve_color_v1 PRIVATE uncertainty_aware_scurve_color_v1)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  foreach(t uncertainty_aware_scurve_color_v1 test_uncertainty_aware_scurve_color_v1)
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endforeach()
endif()
enable_testing()
add_test(NAME uncertainty_aware_scurve_color_v1 COMMAND test_uncertainty_aware_scurve_color_v1)
